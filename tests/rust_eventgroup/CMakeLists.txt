cmake_minimum_required(VERSION 3.20)
project(ove_rust_eventgroup_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(OVE_DIR "../.." ABSOLUTE)
option(OVE_ZERO_HEAP "Use caller-owned EventGroup storage" OFF)
option(OVE_SANITIZE "Instrument the native EventGroup fixture" OFF)
option(OVE_EVENTGROUP_OPTIONAL "Include the legal TIME/ASYNC prerequisites" OFF)
set(config "#ifndef OVE_CONFIG_H\n#define OVE_CONFIG_H\n#define CONFIG_OVE_RTOS_POSIX 1\n#define CONFIG_OVE_EVENTGROUP 1\n")
if(OVE_ZERO_HEAP)
    string(APPEND config "#define CONFIG_OVE_ZERO_HEAP 1\n")
endif()
if(OVE_EVENTGROUP_OPTIONAL)
    string(APPEND config "#define CONFIG_OVE_SYNC 1\n#define CONFIG_OVE_TIME 1\n#define CONFIG_OVE_TIMER 1\n#define CONFIG_OVE_ASYNC 1\n")
endif()
string(APPEND config "#endif\n")
file(WRITE "${CMAKE_BINARY_DIR}/ove_config.h" "${config}")
add_library(ove_eventgroup_test STATIC fixture.c native.c heap.c context.c)
target_include_directories(ove_eventgroup_test PRIVATE "${CMAKE_BINARY_DIR}"
    "${OVE_DIR}/include" "${OVE_DIR}/backends/common" "${OVE_DIR}/tests/host/posix/include")
target_compile_definitions(ove_eventgroup_test PRIVATE _GNU_SOURCE)
target_compile_options(ove_eventgroup_test PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
if(OVE_SANITIZE)
    target_compile_options(ove_eventgroup_test PRIVATE -fsanitize=address,undefined
        -fno-sanitize-recover=all -fno-omit-frame-pointer)
endif()
include("${OVE_DIR}/cmake/OveBindingsCommon.cmake")
_ove_binding_write_sizes_probe("${CMAKE_BINARY_DIR}/ove_storage_sizes.c"
    "#include \"ove/ove.h\"\n#include \"ove/storage.h\"\n")
_ove_binding_build_sizes_probe(eventgroup_sizes ove_eventgroup_test "${CMAKE_BINARY_DIR}/ove_storage_sizes.c")
_ove_binding_extract_sizes("${CMAKE_BINARY_DIR}/ove_storage_sizes.env" eventgroup_sizes
    "Extracting actual POSIX EventGroup storage sizes")
add_custom_target(storage_sizes ALL DEPENDS "${CMAKE_BINARY_DIR}/ove_storage_sizes.env")
