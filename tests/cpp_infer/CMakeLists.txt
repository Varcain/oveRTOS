cmake_minimum_required(VERSION 3.20)
project(ove_cpp_infer_tests CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(OVE_DIR "../.." ABSOLUTE)
option(OVE_SANITIZE "Instrument the recording-only binding fixture" OFF)
foreach(mode heap zero)
 set(config "#pragma once\n#define CONFIG_OVE_RTOS_POSIX 1\n#define CONFIG_OVE_INFER 1\n")
 if(mode STREQUAL "zero")
  string(APPEND config "#define CONFIG_OVE_ZERO_HEAP 1\n")
 endif()
 file(WRITE "${CMAKE_BINARY_DIR}/${mode}/ove_config.h" "${config}")
 foreach(policy hook hook_release default_release default_debug)
  set(target "infer_${mode}_${policy}")
  add_executable(${target} contracts.cpp)
  target_include_directories(${target} BEFORE PRIVATE
   "${CMAKE_BINARY_DIR}/${mode}" "${OVE_DIR}/include"
   "${OVE_DIR}/bindings/cpp" "${OVE_DIR}/tests/host/posix/include")
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  if(policy MATCHES "^default")
   target_compile_definitions(${target} PRIVATE INFER_DEFAULT_ASSERT=1)
  endif()
  if(policy MATCHES "release$")
   target_compile_definitions(${target} PRIVATE NDEBUG=1)
  endif()
  if(OVE_SANITIZE)
   target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
 endforeach()
endforeach()
