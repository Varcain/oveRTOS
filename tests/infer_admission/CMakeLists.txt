cmake_minimum_required(VERSION 3.20)
project(ove_infer_admission_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(OVE_DIR "../.." ABSOLUTE)
option(OVE_ZERO_HEAP "Use caller-owned model storage" OFF)
option(OVE_SANITIZE "Instrument native admission and TFLM" OFF)
set(_config "#ifndef OVE_CONFIG_H\n#define OVE_CONFIG_H\n#define CONFIG_OVE_RTOS_POSIX 1\n#define CONFIG_OVE_INFER 1\n#define CONFIG_OVE_TIME 1\n#define CONFIG_OVE_CONSOLE 1\n")
if(OVE_ZERO_HEAP)
 string(APPEND _config "#define CONFIG_OVE_ZERO_HEAP 1\n")
endif()
string(APPEND _config "#endif\n")
file(WRITE "${CMAKE_BINARY_DIR}/ove_config.h" "${_config}")
if(OVE_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
 add_link_options(-fsanitize=address,undefined)
endif()
set(OVE_DL_DIR "${OVE_DIR}/dl")
set(OVE_GEN_DIR "${CMAKE_BINARY_DIR}")
set(OVE_INCLUDE_DIR "${OVE_DIR}/include")
set(OVE_BACKENDS_COMMON_DIR "${OVE_DIR}/backends/common")
include(${OVE_DIR}/cmake/OveTflm.cmake)
ove_build_tflm()
foreach(_tflm_target ove_tflm ove_tflm_port)
 target_include_directories(${_tflm_target} PRIVATE ${OVE_DIR}/tests/host/posix/include)
endforeach()
add_executable(infer_real real.cc)
target_include_directories(infer_real PRIVATE ${OVE_DIR}/include ${OVE_DIR}/tests/host/posix/include ${CMAKE_BINARY_DIR} ${OVE_DIR}/tests)
target_link_libraries(infer_real PRIVATE ove_tflm m pthread)
add_executable(infer_controlled controlled.cc controlled/arena.cc ${OVE_DIR}/backends/common/ove_infer.cc)
target_include_directories(infer_controlled PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/controlled ${OVE_DIR}/include ${OVE_DIR}/tests/host/posix/include ${CMAKE_BINARY_DIR} ${OVE_DIR}/backends/common)
target_link_libraries(infer_controlled PRIVATE pthread)
target_link_options(infer_controlled PRIVATE -Wl,--wrap=malloc -Wl,--wrap=free)

include(${OVE_DIR}/tests/infer_arena/Targets.cmake)
