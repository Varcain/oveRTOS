cmake_minimum_required(VERSION 3.20)
project(ove_native_boundary_fixture C ASM)
set(OVE_TEST_NATIVE_SOURCE "" CACHE PATH "Previously prepared pinned native source")
set(OVE_TEST_BOARD "" CACHE STRING "Board whose actual linker script is tested")
set(OVE_TEST_IDLE_BYTES 2048 CACHE STRING "Linker-only idle configuration")
set(OVE_TEST_LIVE_BYTES 4 CACHE STRING "Actual fixture BSS object size")
option(CONFIG_ARMV7M_STACKCHECK "Exercise the configured stackcheck reserve" OFF)
if(OVE_TEST_BOARD STREQUAL "qemu-mps2-an500")
    set(OVE_NATIVE_BOARD "mps2-an500")
    set(OVE_NATIVE_BOARD_OPTION MPS2_AN500)
    set(OVE_NATIVE_RAM_START 0x60000000)
    set(OVE_NATIVE_RAM_SIZE 2097152)
    set(ldscript "${OVE_TEST_NATIVE_SOURCE}/boards/arm/mps/mps2-an500/scripts/flash.ld")
    set(NUTTX_CHIP_ABS_DIR "${OVE_TEST_NATIVE_SOURCE}/arch/arm/src/mps")
elseif(OVE_TEST_BOARD STREQUAL "stm32f746g-discovery")
    set(OVE_NATIVE_BOARD "stm32f746g-disco")
    set(OVE_NATIVE_BOARD_OPTION STM32F746G_DISCO)
    set(OVE_NATIVE_RAM_START 0x20010000)
    set(OVE_NATIVE_RAM_SIZE 245760)
    set(ldscript "${OVE_TEST_NATIVE_SOURCE}/boards/arm/stm32f7/stm32f746g-disco/scripts/flash.ld")
    set(NUTTX_CHIP_ABS_DIR "${OVE_TEST_NATIVE_SOURCE}/arch/arm/src/stm32f7")
else()
    message(FATAL_ERROR "Choose a supported native boundary fixture board")
endif()
if(NOT OVE_TEST_IDLE_BYTES MATCHES "^-?[0-9]+$" OR
   NOT OVE_TEST_LIVE_BYTES MATCHES "^(4|8)$")
    message(FATAL_ERROR "Fixture sizes must use the finite decimal test inputs")
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/include/nuttx")
if(CONFIG_ARMV7M_STACKCHECK)
    set(OVE_STACKCHECK_LINE "CONFIG_ARMV7M_STACKCHECK=y")
else()
    set(OVE_STACKCHECK_LINE "# CONFIG_ARMV7M_STACKCHECK is not set")
endif()
configure_file(config.h.in "${CMAKE_BINARY_DIR}/include/nuttx/config.h")
configure_file(config.in "${CMAKE_BINARY_DIR}/fixture.config" @ONLY)

include("${OVE_TEST_NATIVE_SOURCE}/cmake/nuttx_parse_function_args.cmake")
include("${OVE_TEST_NATIVE_SOURCE}/cmake/nuttx_toolchain.cmake")
set(PREPROCESS "${CMAKE_C_COMPILER}" -E -P -x c)
set(LD_SCRIPT_TMP "${CMAKE_BINARY_DIR}/flash.ld.tmp")
file(READ "${OVE_TEST_NATIVE_SOURCE}/CMakeLists.txt" _native_cmake)
string(REGEX MATCHALL
    [=[nuttx_generate_preprocess_target\(SOURCE_FILE \$\{ldscript\} TARGET_FILE[^)]*\$\{LD_SCRIPT_TMP\}[^)]*\)]=]
    _native_calls "${_native_cmake}")
list(LENGTH _native_calls _call_count)
if(NOT _call_count EQUAL 1)
    message(FATAL_ERROR "Expected one actual native linker preprocessing invocation")
endif()
# Execute the real selected invocation, including its actual header dependencies.
# This keeps the incremental checks tied to native integration instead of a copy.
file(WRITE "${CMAKE_BINARY_DIR}/native-preprocess-call.cmake" "${_native_calls}")
include("${CMAKE_BINARY_DIR}/native-preprocess-call.cmake")
add_custom_target(ldscript_tmp DEPENDS "${LD_SCRIPT_TMP}")
add_executable(boundary fixture.S)
set_target_properties(boundary PROPERTIES SUFFIX ".elf" LINKER_LANGUAGE C)
target_include_directories(boundary PRIVATE "${CMAKE_BINARY_DIR}/include")
target_compile_definitions(boundary PRIVATE OVE_TEST_LIVE_BYTES=${OVE_TEST_LIVE_BYTES})
target_compile_options(boundary PRIVATE -mcpu=cortex-m7 -mthumb -mfloat-abi=soft)
target_link_options(boundary PRIVATE -mcpu=cortex-m7 -mthumb -mfloat-abi=soft
    -nostdlib "-Wl,-T,${LD_SCRIPT_TMP},--build-id=none,-Map,${CMAKE_BINARY_DIR}/boundary.map")
set_property(TARGET boundary APPEND PROPERTY LINK_DEPENDS "${LD_SCRIPT_TMP}")
add_dependencies(boundary ldscript_tmp)
